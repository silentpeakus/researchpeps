<script>
  // ── Navbar scroll effect
  const navbar = document.getElementById('navbar');
  window.addEventListener('scroll', () => {
    navbar.classList.toggle('scrolled', window.scrollY > 50);
  });

  // ── Mobile hamburger
  document.getElementById('hamburger').addEventListener('click', function () {
    const links = document.querySelector('.nav-links');
    const isVisible = links.style.display === 'flex';
    links.style.cssText = isVisible
      ? ''
      : 'display:flex;flex-direction:column;gap:16px;position:absolute;top:64px;left:0;right:0;background:rgba(7,13,26,0.97);padding:24px 28px;border-bottom:1px solid rgba(148,163,184,0.09);backdrop-filter:blur(20px)';
  });

  // ── Starfield (lighter than landing — header only)
  (function () {
    const container = document.getElementById('starsEl');
    const frag = document.createDocumentFragment();
    for (let i = 0; i < 45; i++) {
      const el = document.createElement('div');
      el.className = 'star';
      const size = Math.random() > 0.82 ? 2.5 : 1.5;
      el.style.cssText = [
        `left:${(Math.random() * 100).toFixed(1)}%`,
        `top:${(Math.random() * 85).toFixed(1)}%`,
        `width:${size}px`,
        `height:${size}px`,
        `--d:${(2.2 + Math.random() * 3.5).toFixed(1)}s`,
        `--dl:${(Math.random() * 5).toFixed(1)}s`,
        `--lo:${(0.1 + Math.random() * 0.25).toFixed(2)}`,
        `--hi:${(0.55 + Math.random() * 0.45).toFixed(2)}`,
      ].join(';');
      frag.appendChild(el);
    }
    container.appendChild(frag);
  })();

  // ── LIVE WOOCOMMERCE CATALOG
  // Pulls every published product from the store's public Store API
  // and renders it in the landing-page card style. Editing a product
  // in WP admin (image, price, stock, category) updates this page
  // automatically — nothing here is hand-maintained.
  (function () {
    const STORE = 'https://silentpeak.us';
    const grid = document.getElementById('productsGrid');
    const filterBar = document.getElementById('filterBar');
    const countEl = document.getElementById('catalogCount');
    let allProducts = [];

    // Skeleton placeholders while loading
    for (let i = 0; i < 8; i++) {
      const sk = document.createElement('div');
      sk.className = 'product-card skeleton';
      sk.innerHTML = '<div class="product-img-wrap"></div>' +
        '<div class="sk-line"></div><div class="sk-line short"></div>';
      grid.appendChild(sk);
    }

    const formatPrice = (prices) => {
      const minor = Math.pow(10, prices.currency_minor_unit || 2);
      const fmt = (v) => (prices.currency_prefix || prices.currency_symbol || '$') +
        (parseInt(v, 10) / minor).toFixed(prices.currency_minor_unit || 2).replace(/\.00$/, '');
      const onSale = prices.regular_price && prices.regular_price !== prices.price;
      return onSale
        ? '<del>' + fmt(prices.regular_price) + '</del>' + fmt(prices.price)
        : fmt(prices.price);
    };

    const plainText = (html, max) => {
      const div = document.createElement('div');
      div.innerHTML = html || '';
      let text = (div.textContent || '').trim();
      if (max && text.length > max) text = text.slice(0, max).replace(/\s+\S*$/, '') + '…';
      return text;
    };

    const render = (products) => {
      grid.innerHTML = '';
      if (!products.length) {
        grid.innerHTML = '<div class="catalog-message">No compounds found in this category.</div>';
        return;
      }
      products.forEach((p, i) => {
        const card = document.createElement('div');
        card.className = 'product-card';
        const img = (p.images && p.images.length) ? p.images[0] : null;
        const cat = (p.categories && p.categories.length) ? p.categories[0].name : 'Research';
        const tagClass = i % 2 ? 'tag-indigo' : 'tag-blue';
        const desc = plainText(p.short_description || p.description, 150);
        const cartBtn = p.is_in_stock === false
          ? '<span class="btn-cart disabled">Out of Stock</span>'
          : '<a class="btn-cart" href="' + STORE + '/cart/?add-to-cart=' + p.id + '">Add to Cart</a>';

        card.innerHTML =
          '<a class="product-img-wrap" href="' + p.permalink + '">' +
            (img ? '<img src="' + img.src + '" alt="" loading="lazy">' : '') +
          '</a>' +
          '<div class="product-card-body">' +
            '<div class="product-cat"><span class="tag ' + tagClass + '"></span></div>' +
            '<div class="product-name"><a href="' + p.permalink + '"></a></div>' +
            '<p class="product-desc"></p>' +
          '</div>' +
          '<div class="product-card-foot">' +
            '<span class="product-price">' + formatPrice(p.prices) + '</span>' +
            cartBtn +
          '</div>';

        // Text content set safely (names/descriptions may contain entities)
        card.querySelector('.tag').textContent = cat;
        card.querySelector('.product-name a').textContent = plainText(p.name);
        card.querySelector('.product-desc').textContent = desc;
        const imgEl = card.querySelector('img');
        if (imgEl) imgEl.alt = plainText(p.name);
        grid.appendChild(card);
      });
      countEl.textContent = products.length + ' compound' + (products.length === 1 ? '' : 's');
    };

    const buildFilters = () => {
      const cats = new Map();
      allProducts.forEach(p => (p.categories || []).forEach(c => cats.set(c.id, c.name)));
      cats.forEach((name, id) => {
        const chip = document.createElement('button');
        chip.className = 'chip';
        chip.dataset.cat = id;
        chip.textContent = name;
        filterBar.insertBefore(chip, countEl);
      });
      filterBar.addEventListener('click', (e) => {
        const chip = e.target.closest('.chip');
        if (!chip) return;
        filterBar.querySelectorAll('.chip').forEach(c => c.classList.remove('active'));
        chip.classList.add('active');
        const cat = chip.dataset.cat;
        render(cat === '*'
          ? allProducts
          : allProducts.filter(p => (p.categories || []).some(c => String(c.id) === cat)));
      });
    };

    fetch(STORE + '/wp-json/wc/store/v1/products?per_page=100&status=publish')
      .then(res => { if (!res.ok) throw new Error(res.status); return res.json(); })
      .then(products => {
        allProducts = products;
        buildFilters();
        render(allProducts);
      })
      .catch(() => {
        grid.innerHTML = '<div class="catalog-message">The catalog could not be loaded right now.<br>' +
          'You can browse products directly at <a href="' + STORE + '/shop/">silentpeak.us/shop</a>.</div>';
        countEl.textContent = '';
      });
  })();
</script>
