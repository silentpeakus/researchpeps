<script>
  // ── Navbar scroll effect
  const navbar = document.getElementById('navbar');
  window.addEventListener('scroll', () => {
    navbar.classList.toggle('scrolled', window.scrollY > 50);
  });

  // ── Generate starfield
  (function () {
    const container = document.getElementById('starsEl');
    const frag = document.createDocumentFragment();
    for (let i = 0; i < 90; i++) {
      const el = document.createElement('div');
      el.className = 'star';
      const size = Math.random() > 0.82 ? 2.5 : 1.5;
      el.style.cssText = [
        'left:' + (Math.random() * 100).toFixed(1) + '%',
        'top:' + (Math.random() * 68).toFixed(1) + '%',
        'width:' + size + 'px',
        'height:' + size + 'px',
        '--d:' + (2.2 + Math.random() * 3.5).toFixed(1) + 's',
        '--dl:' + (Math.random() * 5).toFixed(1) + 's',
        '--lo:' + (0.1 + Math.random() * 0.25).toFixed(2),
        '--hi:' + (0.55 + Math.random() * 0.45).toFixed(2)
      ].join(';');
      frag.appendChild(el);
    }
    container.appendChild(frag);
  })();

  // ── FAQ accordion
  function toggleFaq(trigger) {
    const item   = trigger.parentElement;
    const answer = item.querySelector('.faq-a');
    const isOpen = answer.classList.contains('open');

    document.querySelectorAll('.faq-a').forEach(a => a.classList.remove('open'));
    document.querySelectorAll('.faq-q').forEach(q => q.classList.remove('open'));

    if (!isOpen) {
      answer.classList.add('open');
      trigger.classList.add('open');
    }
  }

  // ── Mobile hamburger (toggle nav visibility)
  document.getElementById('hamburger').addEventListener('click', function () {
    const links = document.querySelector('.nav-links');
    const isVisible = links.style.display === 'flex';
    links.style.cssText = isVisible
      ? ''
      : 'display:flex;flex-direction:column;gap:16px;position:absolute;top:64px;left:0;right:0;background:rgba(7,13,26,0.97);padding:24px 28px;border-bottom:1px solid rgba(148,163,184,0.09);backdrop-filter:blur(20px)';
  });

  // ── WooCommerce live sync
  // For every .product-card, look the product up in WooCommerce by its
  // slug (data-product-slug) via the public Store API, then fill in:
  //   · the real product image (from the product's featured image)
  //   · the live price (so repricing in WP admin updates this page)
  //   · a working one-click Add to Cart link (/cart/?add-to-cart=ID)
  // If a lookup fails, the card keeps its hand-written price and the
  // button falls back to linking to the product page.
  (function () {
    const STORE = /silentpeak\.us$/.test(location.hostname) ? location.origin : 'https://silentpeak.us';

    const formatPrice = (prices) => {
      const minor = Math.pow(10, prices.currency_minor_unit || 2);
      const value = parseInt(prices.price, 10) / minor;
      const text = value.toFixed(prices.currency_minor_unit || 2).replace(/\.00$/, '');
      return (prices.currency_prefix || prices.currency_symbol || '$') + text;
    };

    document.querySelectorAll('.product-card[data-product-slug]').forEach(async (card) => {
      const slug = card.dataset.productSlug;
      try {
        const res = await fetch(STORE + '/wp-json/wc/store/v1/products?slug=' + encodeURIComponent(slug));
        if (!res.ok) return;
        const matches = await res.json();
        if (!Array.isArray(matches) || matches.length === 0) return;
        const product = matches[0];

        const img = card.querySelector('.product-img-wrap img');
        if (img && product.images && product.images.length) {
          img.src = product.images[0].src;
          img.srcset = product.images[0].srcset || '';
        }

        const priceEl = card.querySelector('.product-price');
        if (priceEl && product.prices) priceEl.textContent = formatPrice(product.prices);

        const btn = card.querySelector('.btn-cart');
        if (btn) {
          if (product.is_in_stock === false) {
            btn.textContent = 'Out of Stock';
            btn.removeAttribute('href');
            btn.style.opacity = '0.45';
            btn.style.pointerEvents = 'none';
          } else {
            // One-click add: WooCommerce adds the item server-side and
            // lands the visitor on the cart page.
            btn.href = STORE + '/cart/?add-to-cart=' + product.id;
          }
        }
      } catch (e) {
        /* network/API unavailable — card keeps its fallback links */
      }
    });
  })();
</script>
